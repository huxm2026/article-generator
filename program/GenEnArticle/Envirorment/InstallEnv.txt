# 一、环境准备（Windows/Linux通用）
# 1. 安装Miniconda（Python环境管理）
# # Linux
# wget https://repo.anaconda.com/miniconda/Miniconda3-latest-Linux-x86_64.sh
# bash Miniconda3-latest-Linux-x86_64.sh

# # Windows
# # 下载安装包：https://docs.conda.io/en/latest/miniconda.html
# a. 下载地址：https://docs.conda.io/en/latest/miniconda.html
#    → 选择 "Windows Installer" + Python 3.x + 64-bit
# b. 双击安装包后：
#    Installation Type → 选 "Just Me"
#    Advanced Options → ✓ Add to PATH（建议勾选）
#    Install Location → 输入 `C:\miniconda3`（强烈推荐默认路径）
# c. 安装完成后，在CMD输入 `conda --version` 验证

# 2. 创建专用环境
# bash
# conda create -n deepseek-1B5 python=3.10.12 
# conda activate deepseek-1B5

# 3. 继承专用环境，安装新的环境
# conda create --name new_env --clone base
# conda activate new_env
# 或：
# conda env export --name base > base_env.yaml
# conda env create --name new_env -f base_env.yaml
# conda activate new_env

# 4.删除指定环境
# conda env list  或 conda info --envs
# conda env remove --name your_env_name 或  conda remove --name your_env_name --all
